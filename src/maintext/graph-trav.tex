\section{WFS遍历框架} \label{sec:wfs}

\nid “爱谁谁优先遍历”\alg{WhateverFirstSearch}（WFS）：
%--
\begin{itemize}
    \item 基本做法：从一个点，顺着边，走到下一个点，直至走遍所有点和边。
    \item 设计要点：
    %--
    \begin{itemize}
        \item 调度：抽象的bag结构（具体设计由后续的设计者来决定）；
        \item 去重：点的二着色。
    \end{itemize}
\end{itemize}

%--------------------
\subsection{WFS的设计}

WFS的设计，如算法\ref{alg:wfs}所示。
WFS的外层调用过程（wrapper）的设计，如算法\ref{alg:wfs-wrapper}所示。

%--
\begin{algorithm}[htbp]
    \caption{WFS($s$)}
    \label{alg:wfs}
    \begin{algorithmic}[1]
        \State put $s$ into the bag $B$
        \While{$B$ is not empty}
        \State take $v$ from $B$
        \If{$v$ is unmarked}
        \State mark $v$
        \ForAll{edge $vw$}
        \State put $w$ into $B$
        \EndFor
        \EndIf
        \EndWhile
    \end{algorithmic}
\end{algorithm}


%--
\begin{algorithm}[htbp]
    \caption{\alg{WFSWrapper}($G$)}
    \label{alg:wfs-wrapper}
    \begin{algorithmic}[1]
        \ForAll{vertices $v$}
        unmark $v$ \Comment{遍历之前，初始化每个点}
        \EndFor
        \ForAll{vertices $v$}
        \If{$v$ is unmarked}
        WFS($v$)
        \EndIf
        \EndFor
    \end{algorithmic}
\end{algorithm}
