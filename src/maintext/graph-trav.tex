\section{图遍历的抽象框架}

\alg{WhateverFirstSearch}（WFS），算法\ref{alg:WFS}。

%--
\begin{algorithm}[htbp]
\caption{\alg{WhateverFirstSearch}($s$)}
\label{alg:WFS}
\begin{algorithmic}[1]
\State put $s$ into the bag $B$
\While{$B$ is not empty}
    \State take $v$ from $B$
    \If{$v$ is unmarked}
        \State mark $v$
        \ForAll{edge $vw$}
            \State put $w$ into $B$
        \EndFor
    \EndIf
\EndWhile
\end{algorithmic}
\end{algorithm}


WFS的外层调用算法（wrapper），算法\ref{alg:WFS-wrapper}。

%--
\begin{algorithm}[htbp]
\caption{\alg{WFSWrapper}($G$)}
\label{alg:WFS-wrapper}
\begin{algorithmic}[1]
\ForAll{vertices $v$}
    unmark $v$
\EndFor
\ForAll{vertices $v$}
    \If{$v$ is unmarked}
        WFS($v$)
    \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}
